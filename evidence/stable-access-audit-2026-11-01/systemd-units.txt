# /home/roma/.config/systemd/user/nozo-relay.service
[Unit]
Description=Nozo Agent Relay (loopback :8765)
After=network-online.target
Wants=network-online.target
StartLimitIntervalSec=120
StartLimitBurst=5

[Service]
Type=simple
WorkingDirectory=/home/roma/Desktop/Nozo Agent Relay/nozo-agent-relay
Environment=HOME=/home/roma
Environment=PATH=/home/roma/.local/bin:/usr/local/bin:/usr/bin:/bin
Environment=NOZO_RELAY_PUBLIC_HOST=roma-gv628rc.tailbfec94.ts.net
Environment=NOZO_RELAY_GATEWAY_ORIGIN=https://romaokorosso.github.io
ExecStart=/bin/bash "/home/roma/Desktop/Nozo Agent Relay/nozo-agent-relay/scripts/run_relay_stable.sh"
Restart=on-failure
RestartSec=5
TimeoutStartSec=60
StandardOutput=append:/home/roma/.local/state/nozo-relay/relay.log
StandardError=append:/home/roma/.local/state/nozo-relay/relay.log

[Install]
WantedBy=default.target
